# v0.34 Facts 查询边界拆分 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将 `kb_agent/facts.py` 中低风险、纯度较高的事实查询与通用工具逻辑拆出，为后续事实抽取流水线拆分打基础，同时保持现有 CLI/API 行为不变。

**Architecture:** `kb_agent/facts.py` 继续作为兼容门面，保留 `extract_facts()` 和现有公开函数导出。新增 `kb_agent/fact_utils.py` 管理事实相关纯工具函数，新增 `kb_agent/fact_queries.py` 管理事实读取、检索、覆盖率与摘要查询。v0.34 不改 LLM 抽取、规则抽取、表格融合、引用融合算法，只移动边界清晰的代码。

**Tech Stack:** Python stdlib, unittest, SQLite-backed project DB helpers, existing CLI integration tests.

---

## File Structure

- Create `kb_agent/fact_utils.py`
  - Owns `query_terms()`, `is_table_source()`, `json_value()`, `normalize_key()`, `confidence()`, and `excerpt()`.
  - Imports only `json`, `re`, typing, and `compact_whitespace` / `unique_strings`.
  - Contains no DB, artifact, LLM, or filesystem access.

- Create `kb_agent/fact_queries.py`
  - Owns public query/read functions: `get_claims()`, `get_entities()`, `get_relations()`, `get_fact_graph()`, `fact_search()`, `fact_coverage_summary()`, and `fact_summary_for_doc()`.
  - Owns query-private helpers: `_search_claim_rows()`, `_search_entity_rows()`, `_search_relation_rows()`, `_like_conditions()`, `_doc_filter()`, `_source_filter()`, `_confidence_filter()`, `_fact_row()`, and `_fact_source_counts()`.
  - Imports `db`, `get_artifact`, and helpers from `fact_utils`.

- Modify `kb_agent/facts.py`
  - Re-export query/read functions from `fact_queries.py` so existing imports keep working.
  - Import utility helpers from `fact_utils.py` under the current private names where internal extraction code still expects them.
  - Remove duplicated query/read/helper function bodies after imports are in place.
  - Keep `extract_facts()`, LLM batch extraction, rule fallback, table/citation merge, artifact construction, and DB replacement in this file for v0.34.

- Modify `kb_agent/quality_baseline.py`
  - Bump `CODE_VERSION` from `v0.33` to `v0.34`.

- Modify `tests/test_ingest_search.py`
  - Update quality-baseline version expectations from `v0.33` to `v0.34`.
  - Keep existing fact extraction/search integration tests unchanged as compatibility regression coverage.

- Create `tests/test_fact_utils.py`
  - Focused unit tests for moved pure helpers.

- Create `tests/test_fact_queries.py`
  - Focused unit tests for fact search, source filtering, confidence filtering, and coverage summary against a temporary SQLite DB.

- Modify `README.md`
  - Add a v0.34 change entry explaining what moved, why it moved, and compatibility impact.

```python
from __future__ import annotations

import unittest

from kb_agent.fact_utils import confidence, excerpt, is_table_source, json_value, normalize_key, query_terms


class FactUtilsTest(unittest.TestCase):
    def test_query_terms_expands_long_chinese_terms(self) -> None:
        terms = query_terms("动态角色任务规划")

        self.assertIn("动态角色任务规划", terms)
        self.assertIn("动态", terms)
        self.assertIn("角色", terms)
        self.assertLessEqual(len(terms), 12)

    def test_json_value_returns_default_for_invalid_json(self) -> None:
        self.assertEqual(json_value("[1, 2]", []), [1, 2])
        self.assertEqual(json_value("not-json", {"fallback": True}), {"fallback": True})

    def test_confidence_clamps_and_rounds(self) -> None:
        self.assertEqual(confidence("1.5", 0.2), 1.0)
        self.assertEqual(confidence("-0.2", 0.2), 0.0)
        self.assertEqual(confidence("0.8766", 0.2), 0.877)
        self.assertEqual(confidence("bad", 0.42), 0.42)

    def test_text_helpers_normalize_table_source_and_excerpt(self) -> None:
        self.assertTrue(is_table_source("llm_table"))
        self.assertFalse(is_table_source("llm_text"))
        self.assertEqual(normalize_key("  Foo， Bar! "), "foobar")
        self.assertEqual(excerpt("a  b  c", 20), "a b c")
        self.assertEqual(excerpt("abcdefghij", 5), "abcde ...")


if __name__ == "__main__":
    unittest.main()
```

```python
from __future__ import annotations

import tempfile
import unittest
from pathlib import Path

from kb_agent import db
from kb_agent.fact_queries import fact_coverage_summary, fact_search


class FactQueriesTest(unittest.TestCase):
    def test_fact_search_filters_by_type_source_and_confidence(self) -> None:
        with tempfile.TemporaryDirectory() as tmp:
            db_path = Path(tmp) / "kb.sqlite"
            conn = db.connect(db_path)
            db.init_db(conn)
            db.insert_paper_claims(
                conn,
                [
                    {
                        "claim_id": "claim_table",
                        "doc_id": "doc_a",
                        "version_id": "v1",
                        "node_id": "node_1",
                        "claim_type": "result",
                        "text": "任务完成率提升 12%",
                        "page_range": [3],
                        "confidence": 0.91,
                        "source": "rule_table",
                        "evidence": {"table_id": "table_001"},
                    },
                    {
                        "claim_id": "claim_text",
                        "doc_id": "doc_a",
                        "version_id": "v1",
                        "node_id": "node_2",
                        "claim_type": "method",
                        "text": "动态角色任务规划用于开放环境",
                        "page_range": [1],
                        "confidence": 0.4,
                        "source": "rule_text",
                        "evidence": {"node_id": "node_2"},
                    },
                ],
            )
            conn.commit()
            conn.close()

            result = fact_search(
                db_path,
                "任务完成率",
                doc_ids=["doc_a"],
                fact_type="claim",
                source="table",
                min_confidence=0.8,
                top_k=5,
            )

        self.assertEqual(result["schema"], "fact_search.v1")
        self.assertEqual(result["count"], 1)
        self.assertEqual(result["items"][0]["fact_id"], "claim_table")
        self.assertEqual(result["items"][0]["source_kind"], "table")

    def test_fact_coverage_summary_counts_table_and_text_sources(self) -> None:
        with tempfile.TemporaryDirectory() as tmp:
            db_path = Path(tmp) / "kb.sqlite"
            conn = db.connect(db_path)
            db.init_db(conn)
            db.insert_paper_claims(
                conn,
                [
                    {
                        "claim_id": "claim_table",
                        "doc_id": "doc_a",
                        "version_id": "v1",
                        "node_id": "node_1",
                        "claim_type": "result",
                        "text": "任务完成率提升 12%",
                        "confidence": 0.91,
                        "source": "rule_table",
                    },
                    {
                        "claim_id": "claim_text",
                        "doc_id": "doc_a",
                        "version_id": "v1",
                        "node_id": "node_2",
                        "claim_type": "method",
                        "text": "动态角色任务规划用于开放环境",
                        "confidence": 0.4,
                        "source": "rule_text",
                    },
                ],
            )
            conn.commit()
            conn.close()

            summary = fact_coverage_summary(db_path, doc_id="doc_a")

        self.assertEqual(summary["schema"], "fact_coverage.v1")
        self.assertEqual(summary["claim_count"], 2)
        self.assertEqual(summary["table_backed_fact_count"], 1)
        self.assertEqual(summary["text_backed_fact_count"], 1)
        self.assertEqual(summary["table_backed_fact_rate"], 0.5)


if __name__ == "__main__":
    unittest.main()
```
